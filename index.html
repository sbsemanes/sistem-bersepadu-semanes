    function formatDateKey(d) {

      const y = d.getFullYear();

      const m = String(d.getMonth() + 1).padStart(2, '0');

      const day = String(d.getDate()).padStart(2, '0');

      return `${y}-${m}-${day}`;

    }

          const a = document.createElement('a');

          a.className = 'dash-card';

          a.href = (modul.link && modul.link !== '') ? modul.link : '#';

          

          let iconHtml = modul.icon.startsWith('http') 

            ? `<img src="${modul.icon}" alt="Icon" />` 

            : `<i class="${modul.icon}"></i>`;


          a.innerHTML = `

            <div class="icon-wrapper">${iconHtml}</div>

            <div class="dash-card-title">${modul.title}</div>

            <div class="dash-card-desc">${modul.desc}</div>

          `;

          dContainer.appendChild(a);

        });

      }


      // Dropdowns Borang

      if (data.dropdown) {

        populateSelect('formNamaGuru', data.dropdown.guru);

        populateSelect('formBilikKhas', data.dropdown.bilikKhas);

        populateSelect('formTahun', data.dropdown.tahun);

        populateSelect('formKelas', data.dropdown.kelas);

      }


      // Kalendar & Senarai Tempahan

      allTempahanData = data.tempahan || [];

      renderCalendar();

      filterBookingsByDate(selectedDateStr);

    }


    function populateSelect(elemId, list) {

      const el = document.getElementById(elemId);

      if (!el || !list) return;

      el.options.length = 1;

      list.forEach(item => {

        const opt = document.createElement('option');

        opt.value = item;

        opt.innerText = item;

        el.appendChild(opt);

      });

    }

    async function handleFormSubmit(e) {

      e.preventDefault();

      const btn = document.getElementById('btnSubmitForm');

      btn.disabled = true;

      btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Menyimpan...';


      const formData = {

        tarikh: document.getElementById('formTarikh').value,

        masaMasuk: document.getElementById('formMasaMasuk').value,

        masaKeluar: document.getElementById('formMasaKeluar').value,

        namaGuru: document.getElementById('formNamaGuru').value,

        tahun: document.getElementById('formTahun').value,

        kelas: document.getElementById('formKelas').value,

        tujuan: document.getElementById('formTujuan').value,

        bilikKhas: document.getElementById('formBilikKhas').value

      };


      try {

        const res = await fetch(APPS_SCRIPT_URL, {

          method: 'POST',

          body: JSON.stringify(formData)

        });

        const result = await res.json();

        

        btn.disabled = false;

        btn.innerHTML = '<i class="fas fa-paper-plane"></i> Hantar Tempahan';


        if (result.success) {

          showToast('Tempahan bilik khas berjaya disimpan!');

          closeBookingModal();

          fetchAllData();

        } else {

          showToast('Ralat: ' + result.message, true);

        }

      } catch (err) {

        btn.disabled = false;

        btn.innerHTML = '<i class="fas fa-paper-plane"></i> Hantar Tempahan';

        showToast('Ralat sambungan: ' + err.toString(), true);

      }

    }


    function renderCalendar() {

      const calGrid = document.getElementById('calendarDays');

      calGrid.innerHTML = '';

      const year = currentCalDate.getFullYear();

      const month = currentCalDate.getMonth();

      const monthNames = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember'];

      document.getElementById('calMonthYear').innerText = `${monthNames[month]} ${year}`;


      const firstDay = new Date(year, month, 1).getDay();

      const totalDays = new Date(year, month + 1, 0).getDate();

      const todayKey = formatDateKey(new Date());


      for (let i = 0; i < firstDay; i++) {

        calGrid.appendChild(document.createElement('div'));

      }


      for (let day = 1; day <= totalDays; day++) {

        const dayDate = new Date(year, month, day);

        const dayKey = formatDateKey(dayDate);


        const dayEl = document.createElement('div');

        dayEl.className = 'cal-day';

        dayEl.innerText = day;


        if (dayKey === todayKey) dayEl.classList.add('today');

        if (dayKey === selectedDateStr) dayEl.classList.add('active');


        const hasBooking = allTempahanData.some(b => b.tarikh === dayKey);

        if (hasBooking) dayEl.classList.add('has-booking');


        dayEl.addEventListener('click', () => {

          document.querySelectorAll('.cal-day').forEach(el => el.classList.remove('active'));

          dayEl.classList.add('active');

          selectedDateStr = dayKey;

          document.getElementById('selectedDateLabel').innerText = `${day} ${monthNames[month]} ${year}`;

          filterBookingsByDate(dayKey);

        });


        calGrid.appendChild(dayEl);

      }

    }


    function prevMonth() { currentCalDate.setMonth(currentCalDate.getMonth() - 1); renderCalendar(); }

    function nextMonth() { currentCalDate.setMonth(currentCalDate.getMonth() + 1); renderCalendar(); }

            Tiada tempahan bilik khas untuk tarikh ini. Bilik sedia digunakan.

      matched.forEach(item => {

          </div>

        container.appendChild(card);

      });

    }
