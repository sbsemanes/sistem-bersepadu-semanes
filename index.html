async function handleFormSubmit(e) { e.preventDefault(); const tarikh = document.getElementById('formTarikh').value.trim(); const bilikKhas = document.getElementById('formBilikKhas').value.trim(); const masaMasuk = document.getElementById('formMasaMasuk').value.trim(); const masaKeluar = document.getElementById('formMasaKeluar').value.trim(); // 1. Validasi: Masa Keluar mesti lebih lewat daripada Masa Masuk if (masaMasuk >= masaKeluar) { showToast("Masa keluar mestilah lebih lewat daripada masa masuk.", true); return; } // Helper format masa ke format HH:mm standard untuk perbandingan tepat const padTime = (t) => { if (!t) return ""; let str = t.toString().trim(); if (str.length === 4 && str.indexOf(":") === 1) str = "0" + str; return str; }; const newStart = padTime(masaMasuk); const newEnd = padTime(masaKeluar); // 2. Semakan Pertindihan Terus di Frontend (Client-side Check) const bertindih = allTempahanData.find(item => { const itemRoom = (item.bilikKhas || "").trim().toLowerCase(); const itemDate = (item.tarikh || "").trim(); const itemStart = padTime(item.masaMasuk); const itemEnd = padTime(item.masaKeluar); // Jika tarikh dan bilik khas adalah sama if (itemRoom === bilikKhas.toLowerCase() && itemDate === tarikh) { // Logik pertindihan: StartA < EndB && EndA > StartB return (newStart < itemEnd && newEnd > itemStart); } return false; }); if (bertindih) { showToast(`Waktu bertindih! ${bilikKhas} telah ditempah oleh ${bertindih.namaGuru || 'guru lain'} (${bertindih.masaMasuk} - ${bertindih.masaKeluar}).`, true); return; } // 3. Hantar ke Backend jika tiada halangan const btn = document.getElementById('btnSubmitForm'); btn.disabled = true; btn.innerHTML = ' Menyimpan...'; const formData = { tarikh: tarikh, masaMasuk: masaMasuk, masaKeluar: masaKeluar, namaGuru: document.getElementById('formNamaGuru').value, tahun: document.getElementById('formTahun').value, kelas: document.getElementById('formKelas').value, tujuan: document.getElementById('formTujuan').value, bilikKhas: bilikKhas }; try { const res = await fetch(APPS_SCRIPT_URL, { method: 'POST', headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ action: "tempahan", formData: formData }) }); const result = await res.json(); btn.disabled = false; btn.innerHTML = ' Hantar Tempahan'; if (result.success) { showToast('Tempahan bilik khas berjaya disimpan!'); closeBookingModal(); fetchAllData(); } else { // Paparkan mesej sekatan waktu daripada Apps Script showToast(result.message || 'Tempahan gagal dibuat.', true); } } catch (err) { btn.disabled = false; btn.innerHTML = ' Hantar Tempahan'; showToast('Ralat sambungan: ' + err.toString(), true); } }
